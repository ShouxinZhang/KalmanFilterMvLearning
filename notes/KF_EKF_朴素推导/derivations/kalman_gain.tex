\section{Kalman 增益如何从创新里长出来}\label{app:kalman-gain}

\subsection{目标}

这一份细节推导只解决一件事：

\begin{quote}
既然更新结构已经定成
\[
\hat{\bm{x}}_{t|t}
= \hat{\bm{x}}_{t|t-1} + \bm{K}_t\tilde{\bm{y}}_t,
\]
那么
\(\bm{K}_t\)
为什么会等于
\[
\bm{P}_{t|t-1}\bm{H}^\top\bm{S}_t^{-1}?
\]
\end{quote}

核心思路不是先写 Joseph 形式，而是顺着下面这条链走：
\begin{enumerate}
  \item 先看创新
  \(\tilde{\bm{y}}_t\)
  里到底混了什么；
  \item 再定义创新自己的不确定性
  \(\bm{S}_t\)；
  \item 再看状态误差和创新的互协方差；
  \item 最后用“更新后误差不该再和创新相关”这一条朴素要求，把
  \(\bm{K}_t\)
  解出来。
\end{enumerate}

\subsection{第一步：创新并不等于真相，它是两部分的混合}

\proofstep{enter}{从创新定义开始：}

\[
\tilde{\bm{y}}_t
= \bm{y}_t - \hat{\bm{y}}_{t|t-1}.
\]

\proofstep{rw}{代入观测方程
\(\bm{y}_t = \bm{H}\bm{x}_t + \bm{v}_t\)
和预测观测
\(\hat{\bm{y}}_{t|t-1}=\bm{H}\hat{\bm{x}}_{t|t-1}\)：}

\[
\tilde{\bm{y}}_t
= \bm{H}\bm{x}_t + \bm{v}_t - \bm{H}\hat{\bm{x}}_{t|t-1}.
\]

\proofstep{rw}{把
\(\bm{x}_t - \hat{\bm{x}}_{t|t-1}\)
认成预测误差
\(\bm{e}_{t|t-1}\)：}

\[
\tilde{\bm{y}}_t
= \bm{H}\bm{e}_{t|t-1} + \bm{v}_t.
\]

\proofstep{done}{所以创新不是“完整的状态误差”，而是“传感器能看见的那部分状态误差”再混上“这一拍的测量噪声”。例如
\(\bm{H}=[1\ 0]\)
时，传感器只直接看见位置误差，看不见速度误差。}

\subsection{第二步：先算创新自己的协方差 \texorpdfstring{$\bm{S}_t$}{S_t}}

\proofstep{enter}{定义创新协方差：}

\[
\bm{S}_t
= \mathbb{E}[\tilde{\bm{y}}_t\tilde{\bm{y}}_t^\top].
\]

\proofstep{rw}{把
\(\tilde{\bm{y}}_t=\bm{H}\bm{e}_{t|t-1}+\bm{v}_t\)
代进去：}

\[
\bm{S}_t
= \mathbb{E}\bigl[(\bm{H}\bm{e}_{t|t-1}+\bm{v}_t)(\bm{H}\bm{e}_{t|t-1}+\bm{v}_t)^\top\bigr].
\]

\proofstep{rw}{按
\((a+b)(a+b)^\top\)
展开：}

\[
\begin{aligned}
\bm{S}_t
=\;&
\mathbb{E}[\bm{H}\bm{e}_{t|t-1}\bm{e}_{t|t-1}^\top\bm{H}^\top] \\
&+ \mathbb{E}[\bm{H}\bm{e}_{t|t-1}\bm{v}_t^\top] \\
&+ \mathbb{E}[\bm{v}_t\bm{e}_{t|t-1}^\top\bm{H}^\top] \\
&+ \mathbb{E}[\bm{v}_t\bm{v}_t^\top].
\end{aligned}
\]

\proofstep{have}{默认预测误差
\(\bm{e}_{t|t-1}\)
和当前测量噪声
\(\bm{v}_t\)
独立，而且
\(\mathbb{E}[\bm{v}_t]=\bm{0}\)，
所以中间两个交叉项为 0。}

\proofstep{rw}{再用
\[
\mathbb{E}[\bm{e}_{t|t-1}\bm{e}_{t|t-1}^\top] = \bm{P}_{t|t-1},
\qquad
\mathbb{E}[\bm{v}_t\bm{v}_t^\top] = \bm{R},
\]
就得到：}

\[
\bm{S}_t
= \bm{H}\bm{P}_{t|t-1}\bm{H}^\top + \bm{R}.
\]

\proofstep{done}{所以
\(\bm{S}_t\)
衡量的是“这一拍创新本身有多不确定”，而不是“状态有多不确定”。}

\subsection{第三步：再算状态误差和创新的互协方差}

\proofstep{enter}{我们关心的是}

\[
\mathbb{E}[\bm{e}_{t|t-1}\tilde{\bm{y}}_t^\top].
\]

\proofstep{rw}{把
\(\tilde{\bm{y}}_t=\bm{H}\bm{e}_{t|t-1}+\bm{v}_t\)
代进去：}

\[
\mathbb{E}[\bm{e}_{t|t-1}\tilde{\bm{y}}_t^\top]
= \mathbb{E}[\bm{e}_{t|t-1}(\bm{H}\bm{e}_{t|t-1}+\bm{v}_t)^\top].
\]

\proofstep{rw}{展开：}

\[
\mathbb{E}[\bm{e}_{t|t-1}\tilde{\bm{y}}_t^\top]
= \mathbb{E}[\bm{e}_{t|t-1}\bm{e}_{t|t-1}^\top]\bm{H}^\top
+ \mathbb{E}[\bm{e}_{t|t-1}\bm{v}_t^\top].
\]

\proofstep{rw}{第二项依然因为独立和零均值而为 0，于是}

\[
\mathbb{E}[\bm{e}_{t|t-1}\tilde{\bm{y}}_t^\top]
= \bm{P}_{t|t-1}\bm{H}^\top.
\]

\proofstep{done}{这条式子在直觉上很重要：如果某个状态方向和创新根本不相关，那创新就没有理由去修它。}

\subsection{第四步：直接把目标函数写出来，对 \texorpdfstring{$\bm{K}_t$}{K_t} 求最小值}

\proofstep{enter}{更新结构已经固定成}

\[
\hat{\bm{x}}_{t|t}
= \hat{\bm{x}}_{t|t-1} + \bm{K}_t\tilde{\bm{y}}_t.
\]

\proofstep{rw}{所以更新后误差就是}

\[
\bm{e}_{t|t}
= \bm{x}_t - \hat{\bm{x}}_{t|t}
= \bm{e}_{t|t-1} - \bm{K}_t\tilde{\bm{y}}_t.
\]

\proofstep{have}{先说朴素想法。既然创新
\(\tilde{\bm{y}}_t\)
已经被拿来修正过一次，那么修完之后的误差里，就不该还残留和这份创新同方向的、可继续利用的信息。}

\proofstep{rw}{但这句话本身还只是直觉。严格证明时，还是应该回到最优化问题本身：直接把目标函数写出来，对
\(\bm{K}_t\)
求最小值。}

\[
J(\bm{K}_t)
= \mathbb{E}[\|\bm{e}_{t|t}\|_2^2]
= \mathrm{tr}(\bm{P}_{t|t}).
\]

\proofstep{rw}{这里最后一个等号来自协方差定义。因为}

\[
\|\bm{e}_{t|t}\|_2^2
= \bm{e}_{t|t}^\top\bm{e}_{t|t}
= \mathrm{tr}(\bm{e}_{t|t}\bm{e}_{t|t}^\top),
\]

\proofstep{rw}{再利用“期望和迹可以交换”，就有}

\[
\mathbb{E}[\|\bm{e}_{t|t}\|_2^2]
= \mathrm{tr}\!\Bigl(
\mathbb{E}[\bm{e}_{t|t}\bm{e}_{t|t}^\top]
\Bigr)
= \mathrm{tr}(\bm{P}_{t|t}).
\]

\proofstep{rw}{如果你还想追问“为什么迹和期望可以交换”，那就把它写成分量和：设
\(\bm{A}=(A_{ij})\)，
则}

\[
\mathrm{tr}(\mathbb{E}[\bm{A}])
= \sum_i \mathbb{E}[A_{ii}]
= \mathbb{E}\Bigl[\sum_i A_{ii}\Bigr]
= \mathbb{E}[\mathrm{tr}(\bm{A})].
\]

\proofstep{rw}{这里中间那一步}

\[
\sum_i \mathbb{E}[A_{ii}]
= \mathbb{E}\Bigl[\sum_i A_{ii}\Bigr]
\]

\proofstep{rw}{用的只是期望的线性性：}

\[
\mathbb{E}[X_1+\cdots+X_n]
= \mathbb{E}[X_1]+\cdots+\mathbb{E}[X_n].
\]

\proofstep{done}{所以这里本质上只是把有限项求和靠期望的线性性移进移出。}

\proofstep{rw}{把
\(\bm{e}_{t|t} = \bm{e}_{t|t-1} - \bm{K}_t\tilde{\bm{y}}_t\)
直接代进去：}

\[
J(\bm{K}_t)
= \mathrm{tr}\!\Bigl(
\mathbb{E}\bigl[
(\bm{e}_{t|t-1} - \bm{K}_t\tilde{\bm{y}}_t)
(\bm{e}_{t|t-1} - \bm{K}_t\tilde{\bm{y}}_t)^\top
\bigr]
\Bigr).
\]

\proofstep{rw}{先把括号展开：}

\[
\begin{aligned}
J(\bm{K}_t)
=\;&
\mathrm{tr}\!\Bigl(
\mathbb{E}[\bm{e}_{t|t-1}\bm{e}_{t|t-1}^\top]
\Bigr) \\
&-
\mathrm{tr}\!\Bigl(
\mathbb{E}[\bm{e}_{t|t-1}\tilde{\bm{y}}_t^\top]\bm{K}_t^\top
\Bigr) \\
&-
\mathrm{tr}\!\Bigl(
\bm{K}_t\mathbb{E}[\tilde{\bm{y}}_t\bm{e}_{t|t-1}^\top]
\Bigr) \\
&+
\mathrm{tr}\!\Bigl(
\bm{K}_t\mathbb{E}[\tilde{\bm{y}}_t\tilde{\bm{y}}_t^\top]\bm{K}_t^\top
\Bigr).
\end{aligned}
\]

\proofstep{rw}{再把前面已经算出来的两个量代入：}

\[
\mathbb{E}[\bm{e}_{t|t-1}\bm{e}_{t|t-1}^\top] = \bm{P}_{t|t-1},
\qquad
\mathbb{E}[\bm{e}_{t|t-1}\tilde{\bm{y}}_t^\top] = \bm{P}_{t|t-1}\bm{H}^\top,
\qquad
\mathbb{E}[\tilde{\bm{y}}_t\tilde{\bm{y}}_t^\top] = \bm{S}_t.
\]

\proofstep{rw}{于是目标函数变成一个显式的二次型：}

\[
\begin{aligned}
J(\bm{K}_t)
=\;&
\mathrm{tr}(\bm{P}_{t|t-1})
- \mathrm{tr}(\bm{P}_{t|t-1}\bm{H}^\top\bm{K}_t^\top) \\
&- \mathrm{tr}(\bm{K}_t\bm{H}\bm{P}_{t|t-1})
+ \mathrm{tr}(\bm{K}_t\bm{S}_t\bm{K}_t^\top).
\end{aligned}
\]

\proofstep{rw}{由于
\(\mathrm{tr}(\bm{P}_{t|t-1}\bm{H}^\top\bm{K}_t^\top)
= \mathrm{tr}(\bm{K}_t\bm{H}\bm{P}_{t|t-1})\)，
前两项可以合并：}

\[
\begin{aligned}
J(\bm{K}_t)
=\;&
\mathrm{tr}(\bm{P}_{t|t-1})
- 2\,\mathrm{tr}(\bm{K}_t\bm{H}\bm{P}_{t|t-1}) \\
&+ \mathrm{tr}(\bm{K}_t\bm{S}_t\bm{K}_t^\top).
\end{aligned}
\]

\proofstep{rw}{现在直接对
\(\bm{K}_t\)
求导。用到两条标准公式：}

\[
\frac{\partial}{\partial \bm{K}}
\mathrm{tr}(\bm{K}\bm{A})
= \bm{A}^\top,
\qquad
\frac{\partial}{\partial \bm{K}}
\mathrm{tr}(\bm{K}\bm{S}\bm{K}^\top)
= 2\bm{K}\bm{S}
\quad(\bm{S}=\bm{S}^\top).
\]

\proofstep{rw}{因此}

\[
\frac{\partial J}{\partial \bm{K}_t}
= -2\bm{P}_{t|t-1}\bm{H}^\top + 2\bm{K}_t\bm{S}_t.
\]

\proofstep{done}{令导数为 0，就得到}

\[
\bm{K}_t
= \bm{P}_{t|t-1}\bm{H}^\top\bm{S}_t^{-1}.
\]

\subsection{一句话理解这条公式}

\[
\bm{K}_t
=
\underbrace{\bm{P}_{t|t-1}\bm{H}^\top}_{\text{创新对状态误差有多有用}}
\underbrace{\bm{S}_t^{-1}}_{\text{创新自己有多可靠}}
\]

它其实就是多维版的“线性回归系数”。一维时更明显：
\[
K_t
= \frac{\mathrm{Cov}(e_{t|t-1},\tilde{y}_t)}{\mathrm{Var}(\tilde{y}_t)}.
\]

所以 Kalman 增益不是神秘公式，而是在回答：

\begin{quote}
这次残差到底有多值得信，以及该把它分配到哪些状态方向上。
\end{quote}

\subsection{工程尾注：如果还要更新协方差}

\proofstep{have}{思想上，前面的推导已经够了。}

\proofstep{rw}{如果实现里还要更新后验协方差，一个常用的数值稳定写法是}

\[
\bm{P}_{t|t}
= (\bm{I} - \bm{K}_t\bm{H})\bm{P}_{t|t-1}(\bm{I} - \bm{K}_t\bm{H})^\top
+ \bm{K}_t\bm{R}\bm{K}_t^\top.
\]

\proofstep{done}{这叫 Joseph 形式。它的角色主要是工程稳健性，不是理解
\(\bm{K}_t\)
公式来龙去脉的主线。}
