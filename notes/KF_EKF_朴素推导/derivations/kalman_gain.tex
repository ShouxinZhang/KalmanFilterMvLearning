\section{目标}

这一份细节推导只解决一件事：

\begin{quote}
既然更新结构已经定成
\[
\hat{\bm{x}}_{t|t}
= \hat{\bm{x}}_{t|t-1} + \bm{K}_t\tilde{\bm{y}}_t,
\]
那么
\(\bm{K}_t\)
为什么会等于
\[
\bm{P}_{t|t-1}\bm{H}^\top\bm{S}_t^{-1}?
\]
\end{quote}

核心思路不是先写 Joseph 形式，而是顺着下面这条链走：
\begin{enumerate}
  \item 先看创新
  \(\tilde{\bm{y}}_t\)
  里到底混了什么；
  \item 再定义创新自己的不确定性
  \(\bm{S}_t\)；
  \item 再看状态误差和创新的互协方差；
  \item 最后用“更新后误差不该再和创新相关”这一条朴素要求，把
  \(\bm{K}_t\)
  解出来。
\end{enumerate}

\section{第一步：创新并不等于真相，它是两部分的混合}

\proofstep{enter}{从创新定义开始：}

\[
\tilde{\bm{y}}_t
= \bm{y}_t - \hat{\bm{y}}_{t|t-1}.
\]

\proofstep{rw}{代入观测方程
\(\bm{y}_t = \bm{H}\bm{x}_t + \bm{v}_t\)
和预测观测
\(\hat{\bm{y}}_{t|t-1}=\bm{H}\hat{\bm{x}}_{t|t-1}\)：}

\[
\tilde{\bm{y}}_t
= \bm{H}\bm{x}_t + \bm{v}_t - \bm{H}\hat{\bm{x}}_{t|t-1}.
\]

\proofstep{rw}{把
\(\bm{x}_t - \hat{\bm{x}}_{t|t-1}\)
认成预测误差
\(\bm{e}_{t|t-1}\)：}

\[
\tilde{\bm{y}}_t
= \bm{H}\bm{e}_{t|t-1} + \bm{v}_t.
\]

\proofstep{done}{所以创新不是“完整的状态误差”，而是“传感器能看见的那部分状态误差”再混上“这一拍的测量噪声”。例如
\(\bm{H}=[1\ 0]\)
时，传感器只直接看见位置误差，看不见速度误差。}

\section{第二步：先算创新自己的协方差 \texorpdfstring{$\bm{S}_t$}{S_t}}

\proofstep{enter}{定义创新协方差：}

\[
\bm{S}_t
= \mathbb{E}[\tilde{\bm{y}}_t\tilde{\bm{y}}_t^\top].
\]

\proofstep{rw}{把
\(\tilde{\bm{y}}_t=\bm{H}\bm{e}_{t|t-1}+\bm{v}_t\)
代进去：}

\[
\bm{S}_t
= \mathbb{E}\bigl[(\bm{H}\bm{e}_{t|t-1}+\bm{v}_t)(\bm{H}\bm{e}_{t|t-1}+\bm{v}_t)^\top\bigr].
\]

\proofstep{rw}{按
\((a+b)(a+b)^\top\)
展开：}

\[
\begin{aligned}
\bm{S}_t
=\;&
\mathbb{E}[\bm{H}\bm{e}_{t|t-1}\bm{e}_{t|t-1}^\top\bm{H}^\top] \\
&+ \mathbb{E}[\bm{H}\bm{e}_{t|t-1}\bm{v}_t^\top] \\
&+ \mathbb{E}[\bm{v}_t\bm{e}_{t|t-1}^\top\bm{H}^\top] \\
&+ \mathbb{E}[\bm{v}_t\bm{v}_t^\top].
\end{aligned}
\]

\proofstep{have}{默认预测误差
\(\bm{e}_{t|t-1}\)
和当前测量噪声
\(\bm{v}_t\)
独立，而且
\(\mathbb{E}[\bm{v}_t]=\bm{0}\)，
所以中间两个交叉项为 0。}

\proofstep{rw}{再用
\[
\mathbb{E}[\bm{e}_{t|t-1}\bm{e}_{t|t-1}^\top] = \bm{P}_{t|t-1},
\qquad
\mathbb{E}[\bm{v}_t\bm{v}_t^\top] = \bm{R},
\]
就得到：}

\[
\bm{S}_t
= \bm{H}\bm{P}_{t|t-1}\bm{H}^\top + \bm{R}.
\]

\proofstep{done}{所以
\(\bm{S}_t\)
衡量的是“这一拍创新本身有多不确定”，而不是“状态有多不确定”。}

\section{第三步：再算状态误差和创新的互协方差}

\proofstep{enter}{我们关心的是}

\[
\mathbb{E}[\bm{e}_{t|t-1}\tilde{\bm{y}}_t^\top].
\]

\proofstep{rw}{把
\(\tilde{\bm{y}}_t=\bm{H}\bm{e}_{t|t-1}+\bm{v}_t\)
代进去：}

\[
\mathbb{E}[\bm{e}_{t|t-1}\tilde{\bm{y}}_t^\top]
= \mathbb{E}[\bm{e}_{t|t-1}(\bm{H}\bm{e}_{t|t-1}+\bm{v}_t)^\top].
\]

\proofstep{rw}{展开：}

\[
\mathbb{E}[\bm{e}_{t|t-1}\tilde{\bm{y}}_t^\top]
= \mathbb{E}[\bm{e}_{t|t-1}\bm{e}_{t|t-1}^\top]\bm{H}^\top
+ \mathbb{E}[\bm{e}_{t|t-1}\bm{v}_t^\top].
\]

\proofstep{rw}{第二项依然因为独立和零均值而为 0，于是}

\[
\mathbb{E}[\bm{e}_{t|t-1}\tilde{\bm{y}}_t^\top]
= \bm{P}_{t|t-1}\bm{H}^\top.
\]

\proofstep{done}{这条式子在直觉上很重要：如果某个状态方向和创新根本不相关，那创新就没有理由去修它。}

\section{第四步：用“更新后误差不该再和创新相关”解出 \texorpdfstring{$\bm{K}_t$}{K_t}}

\proofstep{enter}{更新结构已经固定成}

\[
\hat{\bm{x}}_{t|t}
= \hat{\bm{x}}_{t|t-1} + \bm{K}_t\tilde{\bm{y}}_t.
\]

\proofstep{rw}{所以更新后误差就是}

\[
\bm{e}_{t|t}
= \bm{x}_t - \hat{\bm{x}}_{t|t}
= \bm{e}_{t|t-1} - \bm{K}_t\tilde{\bm{y}}_t.
\]

\proofstep{have}{现在用一个很朴素的最优性要求：既然创新
\(\tilde{\bm{y}}_t\)
已经被用来修正状态了，那么更新后剩下的误差
\(\bm{e}_{t|t}\)
就不该再和创新有系统相关。否则就说明创新里还有没榨干的信息。}

\[
\mathbb{E}[\bm{e}_{t|t}\tilde{\bm{y}}_t^\top] = \bm{0}.
\]

\proofstep{rw}{把
\(\bm{e}_{t|t}=\bm{e}_{t|t-1}-\bm{K}_t\tilde{\bm{y}}_t\)
代进去：}

\[
\mathbb{E}[(\bm{e}_{t|t-1}-\bm{K}_t\tilde{\bm{y}}_t)\tilde{\bm{y}}_t^\top] = \bm{0}.
\]

\proofstep{rw}{展开期望：}

\[
\mathbb{E}[\bm{e}_{t|t-1}\tilde{\bm{y}}_t^\top]
- \bm{K}_t\mathbb{E}[\tilde{\bm{y}}_t\tilde{\bm{y}}_t^\top]
= \bm{0}.
\]

\proofstep{rw}{把上一节算出来的两个量代进去：}

\[
\bm{P}_{t|t-1}\bm{H}^\top - \bm{K}_t\bm{S}_t = \bm{0}.
\]

\proofstep{done}{右乘
\(\bm{S}_t^{-1}\)
就得到}

\[
\bm{K}_t
= \bm{P}_{t|t-1}\bm{H}^\top\bm{S}_t^{-1}.
\]

\section{一句话理解这条公式}

\[
\bm{K}_t
=
\underbrace{\bm{P}_{t|t-1}\bm{H}^\top}_{\text{创新对状态误差有多有用}}
\underbrace{\bm{S}_t^{-1}}_{\text{创新自己有多可靠}}
\]

它其实就是多维版的“线性回归系数”。一维时更明显：
\[
K_t
= \frac{\mathrm{Cov}(e_{t|t-1},\tilde{y}_t)}{\mathrm{Var}(\tilde{y}_t)}.
\]

所以 Kalman 增益不是神秘公式，而是在回答：

\begin{quote}
这次残差到底有多值得信，以及该把它分配到哪些状态方向上。
\end{quote}

\section{工程尾注：如果还要更新协方差}

\proofstep{have}{思想上，前面的推导已经够了。}

\proofstep{rw}{如果实现里还要更新后验协方差，一个常用的数值稳定写法是}

\[
\bm{P}_{t|t}
= (\bm{I} - \bm{K}_t\bm{H})\bm{P}_{t|t-1}(\bm{I} - \bm{K}_t\bm{H})^\top
+ \bm{K}_t\bm{R}\bm{K}_t^\top.
\]

\proofstep{done}{这叫 Joseph 形式。它的角色主要是工程稳健性，不是理解
\(\bm{K}_t\)
公式来龙去脉的主线。}
